);

CREATE TABLE cli_graph_versions (

    CONSTRAINT cli_graph_versions_by_number UNIQUE (graph_id, number),
    PRIMARY KEY (version_id)
);

CREATE TABLE cli_graphs (
    graph_id TEXT NOT NULL,
    project TEXT NOT NULL,
    name TEXT NOT NULL,
    latest_version_id TEXT NOT NULL,

    CONSTRAINT cli_graphs_by_name UNIQUE (project, name),
    PRIMARY KEY (graph_id)
);

CREATE TABLE cli_loop_runs (

);

CREATE TABLE cli_loops (

    inputs TEXT NOT NULL,

    state TEXT NOT NULL,
    pause_reason TEXT DEFAULT '' NOT NULL,
    next_run_at TEXT NOT NULL,
    active_run_id TEXT,
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL,
    CONSTRAINT cli_loops_by_name UNIQUE (project, name),
    PRIMARY KEY (loop_id)
);

CREATE TABLE cli_node_executions (
    execution_id TEXT NOT NULL,
    run_id TEXT NOT NULL,
    node_id TEXT NOT NULL,
    attempt INTEGER NOT NULL,

    started_at TEXT NOT NULL,
    finished_at TEXT,
    error TEXT DEFAULT '' NOT NULL,
    PRIMARY KEY (execution_id)
);

CREATE TABLE cli_run_submissions (
    idempotency_key TEXT NOT NULL,

    PRIMARY KEY (idempotency_key)
);

CREATE TABLE cli_steering (
    sequence INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT,
    steering_id TEXT NOT NULL,

    run_id TEXT NOT NULL,

    message TEXT NOT NULL,
    status TEXT NOT NULL,
    error TEXT DEFAULT '' NOT NULL,
    accepted_at TEXT NOT NULL,
    delivered_at TEXT,
    applied_at TEXT,
    UNIQUE (idempotency_key),
    UNIQUE (steering_id)
);

CREATE TABLE events (

    kind TEXT NOT NULL,

);

CREATE TABLE github_comments (

    posted_at TEXT NOT NULL,

);

    run_id TEXT NOT NULL,
    graph_id TEXT NOT NULL,
    error TEXT DEFAULT '' NOT NULL,
    ordinal INTEGER, auto_approve_nodes TEXT DEFAULT '[]' NOT NULL, runner_overrides TEXT DEFAULT '{}' NOT NULL,
    PRIMARY KEY (run_id)
);

CREATE TABLE sessions (
    run_id TEXT NOT NULL,
    session_key TEXT NOT NULL,
    continuation TEXT NOT NULL,
    PRIMARY KEY (run_id, session_key)
