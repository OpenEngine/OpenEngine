"""Built-in fix graph; composed from shared components, with no alternate entry points."""
from engine.adapters.workspace_provider.git_worktree import GitWorktreeWorkspaceProvider, DEFAULT_ROOT_DIRECTORY
from engine.graph_runtime_langgraph import agent_registry, with_added_agents
from engine.graph_runtime_langgraph.components.change_graphs import Components, fix
from langgraph_acp.providers import CodexACPProvider, ClaudeACPProvider

DEFAULT_RUNNER = "codex"
RUNNERS = (DEFAULT_RUNNER,)

def graph_for(runner, *, session_config=None):
    return fix(Components(runner, with_added_agents(agent_registry([CodexACPProvider(), ClaudeACPProvider()])),
        GitWorktreeWorkspaceProvider(DEFAULT_ROOT_DIRECTORY), session_config))

workflow = graph_for(DEFAULT_RUNNER)
