"""Built-in review graph assembled from shared change components."""
from engine.adapters.agent_runner.acp import installed_codex_env
from engine.adapters.workspace_provider.git_worktree import (
    DEFAULT_ROOT_DIRECTORY, GitWorktreeWorkspaceProvider,
)
from engine.graph_runtime_langgraph import agent_registry, with_added_agents
from engine.graph_runtime_langgraph.components.change_graphs import Components, review
from langgraph_acp.providers import ClaudeACPProvider, CodexACPProvider, OpenCodeACPProvider

DEFAULT_RUNNER = "round-robin"
RUNNERS = (DEFAULT_RUNNER,)
AGENTS = with_added_agents(agent_registry([CodexACPProvider(env=installed_codex_env()), ClaudeACPProvider(), OpenCodeACPProvider()]))


def graph_for(runner=DEFAULT_RUNNER, *, workspace_provider=None, agents=AGENTS, session_config=None):
    return review(Components(runner, agents,
        workspace_provider or GitWorktreeWorkspaceProvider(DEFAULT_ROOT_DIRECTORY), session_config))


workflow = graph_for()
